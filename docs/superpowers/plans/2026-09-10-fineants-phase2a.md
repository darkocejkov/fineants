# fineants Phase 2a Implementation Plan — foundation: timezone, schema, tagging, transfers, watermarks

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the phase 2 data model, a configured timezone with a `local_date` column, categories and ordered rules with manual overrides, automatic transfer pairing, per-connection sync watermarks with error attribution, the tagging pass wired into sync, and the Transactions, Spending, Settings, and Explorer changes that expose all of it.

**Architecture:** Same single Next.js app, library-first. New pure modules under `src/lib/tagging`, `src/lib/time.ts`, `src/lib/sync/attribution.ts`, `src/lib/sync/stale.ts`, `src/lib/db/bootstrap.ts`; new query modules for categories, rules, meta, and connection state; one Drizzle migration creating **every** phase 2 table (including the recurring-cost and debt tables that plan 2b fills in, because `transaction_meta` and `debt_payments` carry foreign keys to them and SQLite cannot add FK columns later). Pages are server components; mutations are server actions; the only client components are the inline category select, the existing Sync button, and charts.

**Tech Stack:** Next.js 16.3 App Router, React 19, TypeScript, Drizzle ORM 0.45 + better-sqlite3 13, drizzle-kit, zod 4, Vitest 5, tsx.

**Spec:** `docs/superpowers/specs/2026-09-10-fineants-phase2-design.md` (sections 1–6, 10–16 for this plan; 7–9 and the Budget/Recurring/Debts pages are plan 2b). Visual system: `docs/superpowers/specs/2026-09-09-fineants-design-direction.md`, binding for all UI tasks.

## Global Constraints

- `src/lib/**` never imports from `next` or `react`; every function takes `db: Db` explicitly.
- Money is integer cents plus a `currency` column. Bank timestamps are Unix seconds. Columns named `*_at` are ISO-8601 text written by fineants. Owner-entered dates and `local_date` are `YYYY-MM-DD` text.
- Effective transaction date is `coalesce(nullif(posted,0), transacted_at, unixepoch(first_seen_at))`; `local_date` is that instant rendered in the configured timezone. Anything grouped by day, week, or month uses `local_date`.
- `TIMEZONE` config defaults to `America/Toronto` and must be a valid IANA zone.
- A `transaction_meta` field whose source is `manual` is never changed by the tagging pass. The pass is idempotent.
- Transfer pairs: different accounts, same currency, equal and opposite amounts, effective dates within 3 days, no pending rows, one partner each, nearest date wins, ties by transaction id.
- Errlist attribution: `conn_id` wins; else exactly one connection whose name (case-insensitive) occurs in `msg`; else unattributed, which holds every watermark.
- Stale `running` sync runs older than 30 minutes are marked `failed` with error `interrupted`.
- UI follows the design direction: ruled ledgers, one 3px ink rule per section, stamps in ink, sentence case, no all-caps, no monospace except raw JSON, no icons, shadows, gradients, badges, pills, middle dots, or appended arrows. Buttons name the outcome ("Save rule").
- Tests: Vitest, in-memory SQLite via `tests/helpers/db.ts`, no network. Run `npm test`, `npx tsc --noEmit`, and `npm run lint` before every commit.

```ts
/**
 * Calendar arithmetic for the app timezone.
 *
 * SQLite knows nothing about DST, so fineants renders every transaction's effective instant to a
 * `local_date` string once, at write time, and groups by that. Everything here is plain
 * TypeScript on Intl; no library.
 */
export const DEFAULT_TIMEZONE = "America/Toronto";

const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

const dateFormatters = new Map<string, Intl.DateTimeFormat>();
const partFormatters = new Map<string, Intl.DateTimeFormat>();

function dateFormatter(tz: string): Intl.DateTimeFormat {
  let f = dateFormatters.get(tz);
  if (!f) {
    // en-CA prints YYYY-MM-DD, which is exactly the storage format.
    f = new Intl.DateTimeFormat("en-CA", { timeZone: tz, year: "numeric", month: "2-digit", day: "2-digit" });
    dateFormatters.set(tz, f);
  }
  return f;
}

function partFormatter(tz: string): Intl.DateTimeFormat {
  let f = partFormatters.get(tz);
  if (!f) {
    f = new Intl.DateTimeFormat("en-CA", {
      timeZone: tz,
      hourCycle: "h23",
      year: "numeric",
      month: "2-digit",
      day: "2-digit",
      hour: "2-digit",
      minute: "2-digit",
      second: "2-digit",
    });
    partFormatters.set(tz, f);
  }
  return f;
}

export function isValidTimeZone(tz: string): boolean {
  if (!tz) return false;
  try {
    new Intl.DateTimeFormat("en-CA", { timeZone: tz });
    return true;
  } catch {
    return false;
  }
}

export function isDateString(s: string): boolean {
  if (!DATE_RE.test(s)) return false;
  const [y, m, d] = s.split("-").map(Number);
  const dt = new Date(Date.UTC(y, m - 1, d));
  return dt.getUTCFullYear() === y && dt.getUTCMonth() === m - 1 && dt.getUTCDate() === d;
}

export function toLocalDate(unixSeconds: number, tz: string): string {
  return dateFormatter(tz).format(new Date(unixSeconds * 1000));
}

export function todayLocal(tz: string, nowMs: number = Date.now()): string {
  return dateFormatter(tz).format(new Date(nowMs));
}

function split(date: string): [number, number, number] {
  const [y, m, d] = date.split("-").map(Number);
  return [y, m, d];
}

function fromUtcMs(ms: number): string {
  return new Date(ms).toISOString().slice(0, 10);
}

/** Calendar days added to a YYYY-MM-DD string; no timezone involved. */
export function addDays(date: string, n: number): string {
  const [y, m, d] = split(date);
  return fromUtcMs(Date.UTC(y, m - 1, d + n));
}

/** b minus a, in whole calendar days (negative when b is earlier). */
export function daysBetween(a: string, b: string): number {
  const [ay, am, ad] = split(a);
  const [by, bm, bd] = split(b);
  return Math.round((Date.UTC(by, bm - 1, bd) - Date.UTC(ay, am - 1, ad)) / 86_400_000);
}

/** The zone's UTC offset at a given instant, in milliseconds (positive east of UTC). */
function offsetMs(utcMs: number, tz: string): number {
  const parts = partFormatter(tz).formatToParts(new Date(utcMs));
  const get = (type: string) => Number(parts.find((p) => p.type === type)?.value ?? "0");
  const asUtc = Date.UTC(get("year"), get("month") - 1, get("day"), get("hour"), get("minute"), get("second"));
  return asUtc - Math.floor(utcMs / 1000) * 1000;
}

/** The instant (ms since epoch) at which the given local date begins in the zone. */
export function localMidnightMs(date: string, tz: string): number {
  const [y, m, d] = split(date);
  const guess = Date.UTC(y, m - 1, d);
  const first = guess - offsetMs(guess, tz);
  // If the offset differs at the candidate instant (we crossed a DST boundary), correct once.
  const second = guess - offsetMs(first, tz);
  return second;
}
```

```ts
import type { ErrItem } from "../simplefin/types";

export type Attribution = { items: ErrItem[]; heldAll: boolean; heldIds: Set<string> };

/**
 * Decide which connection each errlist item is about. The new wire shape says so with
 * `conn_id`; the old shape only has prose, so the connection whose name appears in it wins,
 * but only when exactly one does. Anything undecidable holds every watermark for this run:
 * re-fetching a window is cheap and idempotent, skipping one is silent data loss.
 */
export function attributeErrors(errlist: ErrItem[], connections: { id: string; name: string }[]): Attribution {
  const ids = new Set(connections.map((c) => c.id));
  const items: ErrItem[] = [];
  const heldIds = new Set<string>();
  let heldAll = false;

  for (const e of errlist) {
    let attributed: string | undefined;
    if (e.connId && ids.has(e.connId)) {
      attributed = e.connId;
    } else if (!e.connId) {
      const msg = e.msg.toLowerCase();
      const named = connections.filter((c) => c.name.trim() && msg.includes(c.name.trim().toLowerCase()));
      if (named.length === 1) attributed = named[0].id;
    }
    if (attributed) {
      heldIds.add(attributed);
      items.push({ ...e, attributedConnectionId: attributed });
    } else {
      heldAll = true;
      items.push({ ...e });
    }
  }
  if (heldAll) for (const id of ids) heldIds.add(id);
  return { items, heldAll, heldIds };
}
```

```ts
import { and, eq, lt } from "drizzle-orm";
import type { Db } from "../db/client";
import { syncRuns } from "../db/schema";

export const STALE_RUN_SECONDS = 30 * 60;

/**
 * A run row is inserted before its HTTP request; a crash between the two leaves it `running`
 * forever, which the overview reads as "still running". Anything that old is dead.
 */
export function sweepStaleRuns(db: Db, nowMs: number): number {
  const cutoff = new Date(nowMs - STALE_RUN_SECONDS * 1000).toISOString();
  return db
    .update(syncRuns)
    .set({ status: "failed", finishedAt: new Date(nowMs).toISOString(), error: "interrupted" })
    .where(and(eq(syncRuns.status, "running"), lt(syncRuns.startedAt, cutoff)))
    .run().changes;
}
```

```ts
import { eq } from "drizzle-orm";
import type { Db } from "../db/client";
import { connectionSyncState, type ConnectionSyncStateRow } from "../db/schema";

type StateDb = Pick<Db, "select" | "insert" | "update">;

export function getConnectionStates(db: StateDb): ConnectionSyncStateRow[] {
  return db.select().from(connectionSyncState).orderBy(connectionSyncState.connectionId).all();
}

/**
 * The window start every sync must reach back to.
 * `undefined`: no connection has a row yet (fall back to the global watermark).
 * `null`: some connection has never had a clean run (full backfill).
 * number: the oldest clean end across connections.
 */
export function oldestWatermark(db: StateDb): number | null | undefined {
  const rows = getConnectionStates(db);
  if (rows.length === 0) return undefined;
  let min: number | null = Number.POSITIVE_INFINITY;
  for (const r of rows) {
    if (r.lastSuccessEnd === null) return null;
    if (r.lastSuccessEnd < min!) min = r.lastSuccessEnd;
  }
  return min;
}

export function ensureConnectionState(db: StateDb, connectionId: string, seedEnd: number | null): void {
  db.insert(connectionSyncState).values({ connectionId, lastSuccessEnd: seedEnd }).onConflictDoNothing().run();
}

export function advanceWatermark(db: StateDb, connectionId: string, end: number): void {
  db.update(connectionSyncState)
    .set({ lastSuccessEnd: end, lastErrorCode: null, lastErrorMsg: null, lastErrorAt: null })
    .where(eq(connectionSyncState.connectionId, connectionId))
    .run();
}

export function recordConnectionError(db: StateDb, connectionId: string, code: string, msg: string, atIso: string): void {
  db.update(connectionSyncState)
    .set({ lastErrorCode: code, lastErrorMsg: msg, lastErrorAt: atIso })
    .where(eq(connectionSyncState.connectionId, connectionId))
    .run();
}
```

```ts
export function listCategories(db: Db, opts: { includeArchived?: boolean } = {}): CategoryRow[] {
  const q = db.select().from(categories);

function assertNameFree(db: Db, name: string, exceptId?: number): void {
  const clash = db.select({ id: categories.id }).from(categories).where(sql`lower(${categories.name}) = lower(${name})`).get();
  if (clash && clash.id !== exceptId) throw new CategoryError(`A category named "${name}" already exists.`);

export function createCategory(db: Db, input: { name: string; kind: CategoryKind }, nowIso = now()): CategoryRow {
  const name = input.name.trim();
  if (!name) throw new CategoryError("A category needs a name.");
  assertNameFree(db, name);
  const max = db.select({ m: sql<number>`coalesce(max(${categories.sortOrder}), -1)`.mapWith(Number) }).from(categories).get()!.m;
  return db.insert(categories).values({ name, kind: input.kind, sortOrder: max + 1, createdAt: nowIso, updatedAt: nowIso }).returning().get();
}

  db: Db,
  id: number,
  patch: { name?: string; kind?: CategoryKind; monthlyCapCents?: number | null; capCurrency?: string | null },
  nowIso = now(),
): void {
  const set: Partial<typeof categories.$inferInsert> = { updatedAt: nowIso };
  if (patch.name !== undefined) {
    const name = patch.name.trim();
    if (!name) throw new CategoryError("A category needs a name.");
    assertNameFree(db, name, id);
    set.name = name;
  }
  if (patch.kind !== undefined) set.kind = patch.kind;
  if (patch.monthlyCapCents !== undefined) {
    if (patch.monthlyCapCents !== null && (!Number.isInteger(patch.monthlyCapCents) || patch.monthlyCapCents <= 0)) {
      throw new CategoryError("A monthly cap must be a positive amount.");
    }
    set.monthlyCapCents = patch.monthlyCapCents;
    if (patch.monthlyCapCents === null) set.capCurrency = null;
  }
  if (patch.capCurrency !== undefined) set.capCurrency = patch.capCurrency;
  db.update(categories).set(set).where(eq(categories.id, id)).run();
}

  db.update(categories).set({ archived, updatedAt: nowIso }).where(eq(categories.id, id)).run();

export function moveCategory(db: Db, id: number, direction: "up" | "down"): void {
  const all = listCategories(db, { includeArchived: true });
  const i = all.findIndex((c) => c.id === id);
  if (i < 0) return;
  const j = direction === "up" ? i - 1 : i + 1;
```

```ts
import { asc, eq, sql } from "drizzle-orm";
import type { Db } from "../db/client";
import { rules, RULE_FIELDS, RULE_MATCHES, type RuleField, type RuleMatch, type RuleRow } from "../db/schema";

export type RuleInput = {
  name?: string;
  field: RuleField;
  match: RuleMatch;
  pattern: string;
  accountId?: string | null;
  minAmountCents?: number | null;
  maxAmountCents?: number | null;
  categoryId?: number | null;
  markTransfer?: boolean;
  active?: boolean;
};

export class RuleError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "RuleError";
  }
}

const now = () => new Date().toISOString();

/**
 * Returns a message for the first problem, or null when the input is usable.
 * `forPreview` skips the "must set something" check: a preview only needs a matchable pattern.
 */
export function validateRuleInput(input: RuleInput, opts: { forPreview?: boolean } = {}): string | null {
  if (!(RULE_FIELDS as readonly string[]).includes(input.field)) return "Choose which field the rule looks at.";
  if (!(RULE_MATCHES as readonly string[]).includes(input.match)) return "Choose how the pattern matches.";
  const pattern = input.pattern.trim();
  if (!pattern) return "A rule needs a pattern to look for.";
  if (input.match === "regex") {
    try {
      new RegExp(pattern, "i");
    } catch (e) {
      return `That is not a valid regular expression: ${e instanceof Error ? e.message : String(e)}`;
    }
  }
  if (!opts.forPreview && (input.categoryId ?? null) === null && !input.markTransfer) return "A rule must set a category or mark transfers, or both.";
  const min = input.minAmountCents ?? null;
  const max = input.maxAmountCents ?? null;
  if (min !== null && !Number.isInteger(min)) return "The minimum amount must be a whole number of cents.";
  if (max !== null && !Number.isInteger(max)) return "The maximum amount must be a whole number of cents.";
  if (min !== null && max !== null && min > max) return "The minimum amount is larger than the maximum.";
  return null;
}

export function listRules(db: Db, opts: { includeInactive?: boolean } = {}): RuleRow[] {
  const q = db.select().from(rules);
  return (opts.includeInactive ? q : q.where(eq(rules.active, true))).orderBy(asc(rules.priority), asc(rules.id)).all();
}

export function getRule(db: Db, id: number): RuleRow | undefined {
  return db.select().from(rules).where(eq(rules.id, id)).get();
}

function toRow(input: RuleInput, nowIso: string) {
  const pattern = input.pattern.trim();
  return {
    name: (input.name ?? "").trim() || pattern,
    field: input.field,
    match: input.match,
    pattern,
    accountId: input.accountId ?? null,
    minAmountCents: input.minAmountCents ?? null,
    maxAmountCents: input.maxAmountCents ?? null,
    categoryId: input.categoryId ?? null,
    markTransfer: input.markTransfer ?? false,
    active: input.active ?? true,
    updatedAt: nowIso,
  };
}

export function createRule(db: Db, input: RuleInput, nowIso = now()): RuleRow {
  const problem = validateRuleInput(input);
  if (problem) throw new RuleError(problem);
  const max = db.select({ m: sql<number>`coalesce(max(${rules.priority}), -1)`.mapWith(Number) }).from(rules).get()!.m;
  return db.insert(rules).values({ ...toRow(input, nowIso), priority: max + 1, createdAt: nowIso }).returning().get();
}

export function updateRule(db: Db, id: number, input: RuleInput, nowIso = now()): void {
  const problem = validateRuleInput(input);
  if (problem) throw new RuleError(problem);
  db.update(rules).set(toRow(input, nowIso)).where(eq(rules.id, id)).run();
}

export function deleteRule(db: Db, id: number): void {
  db.delete(rules).where(eq(rules.id, id)).run();
}

export function moveRule(db: Db, id: number, direction: "up" | "down"): void {
  const all = listRules(db, { includeInactive: true });
  const i = all.findIndex((r) => r.id === id);
  if (i < 0) return;
  const j = direction === "up" ? i - 1 : i + 1;
  if (j < 0 || j >= all.length) return;
  db.transaction((tx) => {
    all.forEach((r, k) => tx.update(rules).set({ priority: k }).where(eq(rules.id, r.id)).run());
    tx.update(rules).set({ priority: j }).where(eq(rules.id, all[i].id)).run();
    tx.update(rules).set({ priority: i }).where(eq(rules.id, all[j].id)).run();
  });
}
```

```ts
import type { RuleRow } from "../db/schema";

export type TxLike = { id: string; accountId: string; amountCents: number; payee: string | null; description: string };
export type RuleLike = Pick<RuleRow, "id" | "field" | "match" | "pattern" | "accountId" | "minAmountCents" | "maxAmountCents" | "categoryId" | "markTransfer">;
export type CompiledRule = RuleLike & { test(t: TxLike): boolean };

function textOf(t: TxLike, field: RuleLike["field"]): string[] {
  if (field === "payee") return t.payee ? [t.payee] : [];
  if (field === "description") return [t.description];
  return t.payee ? [t.payee, t.description] : [t.description];
}

/** Rules in the given order, minus any whose regex does not compile (reported, not thrown). */
export function compileRules(rules: RuleLike[]): { compiled: CompiledRule[]; invalid: { id: number; error: string }[] } {
  const compiled: CompiledRule[] = [];
  const invalid: { id: number; error: string }[] = [];
  for (const r of rules) {
    let matcher: (s: string) => boolean;
    if (r.match === "regex") {
      try {
        const re = new RegExp(r.pattern, "i");
        matcher = (s) => re.test(s);
      } catch (e) {
        invalid.push({ id: r.id, error: e instanceof Error ? e.message : String(e) });
        continue;
      }
    } else {
      const needle = r.pattern.toLowerCase();
      matcher = (s) => s.toLowerCase().includes(needle);
    }
    compiled.push({
      ...r,
      test(t) {
        if (r.accountId && t.accountId !== r.accountId) return false;
        if (r.minAmountCents !== null && t.amountCents < r.minAmountCents) return false;
        if (r.maxAmountCents !== null && t.amountCents > r.maxAmountCents) return false;
        return textOf(t, r.field).some(matcher);
      },
    });
  }
  return { compiled, invalid };
}

/** First matching rule that sets a category, and first that marks transfers; independently. */
export function firstMatches(compiled: CompiledRule[], t: TxLike): { category: CompiledRule | null; transfer: CompiledRule | null } {
  let category: CompiledRule | null = null;
  let transfer: CompiledRule | null = null;
  for (const r of compiled) {
    if (category && transfer) break;
    if ((!category && r.categoryId !== null) || (!transfer && r.markTransfer)) {
      if (!r.test(t)) continue;
      if (!category && r.categoryId !== null) category = r;
      if (!transfer && r.markTransfer) transfer = r;
    }
  }
  return { category, transfer };
}
```

```ts
export type TransferCandidate = { id: string; accountId: string; currency: string; amountCents: number; dateSec: number; pending: boolean };
export type TransferPair = { outflowId: string; inflowId: string };

export const TRANSFER_WINDOW_DAYS = 3;

/**
 * Money that leaves one of the owner's accounts and arrives in another, same currency, same
 * amount, within a few days, is a transfer. Greedy: outflows in date order each take the
 * nearest free inflow; ties on distance go to the smaller id so the result is deterministic.
 */
export function pairTransfers(rows: TransferCandidate[], maxDays = TRANSFER_WINDOW_DAYS): TransferPair[] {
  const maxSec = maxDays * 86400;
  const groups = new Map<string, { outs: TransferCandidate[]; ins: TransferCandidate[] }>();
  for (const r of rows) {
    if (r.pending || r.amountCents === 0) continue;
    const key = `${r.currency}|${Math.abs(r.amountCents)}`;
    let g = groups.get(key);
    if (!g) groups.set(key, (g = { outs: [], ins: [] }));
    (r.amountCents < 0 ? g.outs : g.ins).push(r);
  }
  const byDateThenId = (a: TransferCandidate, b: TransferCandidate) => a.dateSec - b.dateSec || (a.id < b.id ? -1 : a.id > b.id ? 1 : 0);
  const pairs: TransferPair[] = [];
  for (const g of groups.values()) {
    if (g.outs.length === 0 || g.ins.length === 0) continue;
    g.outs.sort(byDateThenId);
    const taken = new Set<string>();
    for (const out of g.outs) {
      let best: TransferCandidate | null = null;
      let bestDist = Number.POSITIVE_INFINITY;
      for (const inn of g.ins) {
        if (taken.has(inn.id) || inn.accountId === out.accountId) continue;
        const dist = Math.abs(inn.dateSec - out.dateSec);
        if (dist > maxSec) continue;
        if (dist < bestDist || (dist === bestDist && best && inn.id < best.id)) {
          best = inn;
          bestDist = dist;
        }
      }
      if (best) {
        taken.add(best.id);
        pairs.push({ outflowId: out.id, inflowId: best.id });
      }
    }
  }
  return pairs;
}
```

```ts
import { eq } from "drizzle-orm";
import type { Db } from "../db/client";
import { transactionMeta, type TransactionMetaRow } from "../db/schema";

const now = () => new Date().toISOString();

export function getMeta(db: Db, transactionId: string): TransactionMetaRow | undefined {
  return db.select().from(transactionMeta).where(eq(transactionMeta.transactionId, transactionId)).get();
}

type MetaPatch = Partial<Omit<TransactionMetaRow, "transactionId">>;

/** Upsert a partial meta row. The pass owns deletion of empty rows; manual writes never delete. */
export function upsertMeta(db: Pick<Db, "insert">, transactionId: string, patch: MetaPatch, nowIso = now()): void {
  db.insert(transactionMeta)
    .values({ transactionId, updatedAt: nowIso, ...patch })
    .onConflictDoUpdate({ target: transactionMeta.transactionId, set: { ...patch, updatedAt: nowIso } })
    .run();
}

/** Owner picks a category (or "none"). Pinned against the pass until cleared. */
export function setCategory(db: Db, transactionId: string, categoryId: number | null, nowIso = now()): void {
  upsertMeta(db, transactionId, { categoryId, categorySource: "manual", ruleId: null }, nowIso);
}

/** Back to whatever the rules say; the caller (server action) or applyRulesToOne re-derives it. */
export function clearCategoryOverride(db: Db, transactionId: string, nowIso = now()): void {
  const m = getMeta(db, transactionId);
  if (!m || m.categorySource !== "manual") return;
  upsertMeta(db, transactionId, { categoryId: null, categorySource: null, ruleId: null }, nowIso);
  // Local import to avoid a module cycle at load time (apply.ts imports this file).
  const { applyRulesToOne } = require("../tagging/apply") as typeof import("../tagging/apply");
  applyRulesToOne(db, transactionId);
}

export function setTransfer(db: Db, transactionId: string, isTransfer: boolean, nowIso = now()): void {
  const m = getMeta(db, transactionId);
  db.transaction((tx) => {
    // Free an auto partner so the next pass can re-pair it with something else.
    if (m?.counterpartTransactionId && m.transferSource === "auto") {
      upsertMeta(tx, m.counterpartTransactionId, { isTransfer: false, transferSource: null, counterpartTransactionId: null }, nowIso);
    }
    upsertMeta(tx, transactionId, { isTransfer, transferSource: "manual", counterpartTransactionId: isTransfer ? (m?.counterpartTransactionId ?? null) : null }, nowIso);
  });
}

export function setRecurring(db: Db, transactionId: string, recurringCostId: number | null, nowIso = now()): void {
  upsertMeta(db, transactionId, { recurringCostId, recurringSource: "manual" }, nowIso);
}
```

```ts
  categoryCleared: number;
  transfersPaired: number;
  transfersCleared: number;
  recurringMatched: number;
  invalidRules: { id: number; error: string }[];

  hidden: boolean;

  description: string;
  pending: boolean;

function loadAll(db: Db, ids?: string[]): Loaded[] {

    .select({
      id: transactions.id,
      accountId: transactions.accountId,
      currency: accounts.currency,
      hidden: accounts.hidden,

  let cleared = false;
  const m = firstMatches(compiled, row);
  if (d.categorySource !== "manual") {
    const next = m.category ? m.category.categoryId : null;
    const nextRule = m.category ? m.category.id : null;
    if (next !== d.categoryId || nextRule !== d.ruleId) {
      if (next === null && d.categoryId !== null) cleared = true;
      if (next !== null && next !== d.categoryId) set = true;
      d.categoryId = next;
      d.ruleId = nextRule;
      d.categorySource = next === null ? null : "rule";
    }
  }
  if (d.transferSource !== "manual") {
    if (m.transfer) {
      d.isTransfer = true;
      d.transferSource = "rule";
      d.counterpartTransactionId = null;
    } else if (d.transferSource === "rule") {
      d.isTransfer = false;
      d.transferSource = null;
    }
  }
  return { set, cleared };
}

export function applyTagging(db: Db, opts: { now?: () => number } = {}): TaggingReport {
  const now = opts.now ?? Date.now;
  const started = now();
  const { compiled, invalid } = compileRules(listRules(db));
  const report: TaggingReport = { categorised: 0, categoryCleared: 0, transfersPaired: 0, transfersCleared: 0, recurringMatched: 0, invalidRules: invalid, durationMs: 0, at: new Date(started).toISOString() };

  db.transaction((tx) => {
    const rows = loadAll(tx as unknown as Db);
    const desired = new Map<string, Desired>();
    const before = new Map<string, Desired>();

    // 6.1 rules
    for (const row of rows) {
      const d = fromMeta(row.meta);
      before.set(row.id, { ...d });
      const r = applyRules(compiled, row, d);
      if (r.set) report.categorised++;
      if (r.cleared) report.categoryCleared++;
      desired.set(row.id, d);
    }

    // 6.2 transfers: drop every auto pairing, then recompute among rows nobody pinned.
    for (const [, d] of desired) {
      if (d.transferSource === "auto") {
        d.isTransfer = false;
        d.transferSource = null;
        d.counterpartTransactionId = null;
      }
    }
    const candidates = rows.filter((r) => {
      const d = desired.get(r.id)!;
      return !r.hidden && d.transferSource !== "manual" && d.transferSource !== "rule";
    });
    const pairs = pairTransfers(candidates.map((r) => ({ id: r.id, accountId: r.accountId, currency: r.currency, amountCents: r.amountCents, dateSec: r.dateSec, pending: r.pending })));
    for (const p of pairs) {
      const a = desired.get(p.outflowId)!;
      const b = desired.get(p.inflowId)!;
      a.isTransfer = true; a.transferSource = "auto"; a.counterpartTransactionId = p.inflowId;
      b.isTransfer = true; b.transferSource = "auto"; b.counterpartTransactionId = p.outflowId;
    }

    // 6.3 recurring: filled in by plan 2b. recurringMatched stays 0 here.

    // Write only what changed.
    const stamp = new Date(now()).toISOString();
    let newlyPairedRows = 0;
    for (const row of rows) {
      const d = desired.get(row.id)!;
      const b = before.get(row.id)!;
      if (b.isTransfer && !d.isTransfer) report.transfersCleared++;
      if (!b.isTransfer && d.isTransfer && d.transferSource === "auto") newlyPairedRows++;
      if (same(b, d)) continue;
      if (isEmpty(d)) {
        if (row.meta) tx.delete(transactionMeta).where(eq(transactionMeta.transactionId, row.id)).run();
        continue;
      }
      tx.insert(transactionMeta)
        .values({ transactionId: row.id, ...d, updatedAt: stamp })
        .onConflictDoUpdate({ target: transactionMeta.transactionId, set: { ...d, updatedAt: stamp } })
        .run();
    }
    // A pair is two rows; an odd count means one side was already auto (re-paired to a new partner).
    report.transfersPaired = Math.ceil(newlyPairedRows / 2);
  });

  report.durationMs = now() - started;
  setState(db, STATE_KEYS.lastTagging, report);
  return report;
}

/** Step 6.1 for a single transaction, used after a manual override is cleared. */
export function applyRulesToOne(db: Db, transactionId: string): void {
  const { compiled } = compileRules(listRules(db));
  const row = loadAll(db, [transactionId])[0];
  if (!row) return;
  const d = fromMeta(row.meta);
  applyRules(compiled, row, d);
  if (isEmpty(d)) {
    if (row.meta) db.delete(transactionMeta).where(eq(transactionMeta.transactionId, row.id)).run();
    return;
  }
  const stamp = new Date().toISOString();
  db.insert(transactionMeta)
    .values({ transactionId: row.id, ...d, updatedAt: stamp })
    .onConflictDoUpdate({ target: transactionMeta.transactionId, set: { ...d, updatedAt: stamp } })
    .run();
}

export function getLastTagging(db: Db): TaggingReport | undefined {
  return getState<TaggingReport>(db, STATE_KEYS.lastTagging);
}
```

```ts
import { eq, isNull } from "drizzle-orm";
import type { Db } from "./client";
import { connections, connectionSyncState, transactions } from "./schema";
import { getState, setState, STATE_KEYS } from "./appState";
import { toLocalDate } from "../time";
import { seedDefaultCategories } from "../queries/categories";
import { ensureConnectionState } from "../queries/connectionState";
import { applyTagging } from "../tagging/apply";
import { effectiveDate } from "../queries/transactions";

export const BOOTSTRAP_VERSION = 1;

export type BootstrapReport = {
  localDatesBackfilled: number;
  categoriesSeeded: number;
  connectionsSeeded: number;
  timezoneRecomputed: boolean;
  version: number;
};

function writeLocalDates(db: Db, tz: string, onlyNull: boolean): number {
  const q = db.select({ id: transactions.id, sec: effectiveDate }).from(transactions);
  const rows = onlyNull ? q.where(isNull(transactions.localDate)).all() : q.all();
  db.transaction((tx) => {
    for (const r of rows) tx.update(transactions).set({ localDate: toLocalDate(r.sec, tz) }).where(eq(transactions.id, r.id)).run();
  });
  return rows.length;
}

/**
 * Data migrations that SQL alone cannot express, run after the schema migration. Each step is
 * idempotent; the version gate just keeps a healthy boot from re-scanning tables for nothing.
 */
export function runDataSteps(db: Db, opts: { timezone: string; now?: () => number }): BootstrapReport {
  const nowIso = new Date((opts.now ?? Date.now)()).toISOString();
  const report: BootstrapReport = { localDatesBackfilled: 0, categoriesSeeded: 0, connectionsSeeded: 0, timezoneRecomputed: false, version: BOOTSTRAP_VERSION };
  const storedTz = getState<string>(db, STATE_KEYS.timezone);
  const version = getState<number>(db, STATE_KEYS.bootstrapVersion) ?? 0;

  if (version < 1) {
    report.categoriesSeeded = seedDefaultCategories(db, nowIso);
    const seedEnd = getState<number>(db, STATE_KEYS.lastSuccessfulSyncEnd) ?? null;
    const known = new Set(db.select({ id: connectionSyncState.connectionId }).from(connectionSyncState).all().map((r) => r.id));
    for (const c of db.select({ id: connections.id }).from(connections).all()) {
      if (known.has(c.id)) continue;
      ensureConnectionState(db, c.id, seedEnd);
      report.connectionsSeeded++;
    }
    setState(db, STATE_KEYS.bootstrapVersion, 1);
  }

  if (storedTz !== undefined && storedTz !== opts.timezone) {
    report.localDatesBackfilled = writeLocalDates(db, opts.timezone, false);
    report.timezoneRecomputed = true;
    setState(db, STATE_KEYS.timezone, opts.timezone);
    applyTagging(db, { now: opts.now });
  } else {
    report.localDatesBackfilled = writeLocalDates(db, opts.timezone, true);
    if (storedTz === undefined) setState(db, STATE_KEYS.timezone, opts.timezone);
  }
  return report;
}
```

```ts
export type PeriodTotals = { period: string; currency: string; outflowCents: number; inflowCents: number };

function periodExpr(g: Granularity) {
  if (g === "day") return sql<string>`${transactions.localDate}`;
  if (g === "month") return sql<string>`substr(${transactions.localDate}, 1, 7)`;
  return sql<string>`strftime('%G-W%V', ${transactions.localDate})`;
}

const outflow = sql<number>`sum(case when ${transactions.amountCents} < 0 then -${transactions.amountCents} else 0 end)`.mapWith(Number);
const inflow = sql<number>`sum(case when ${transactions.amountCents} > 0 then ${transactions.amountCents} else 0 end)`.mapWith(Number);

/** Non-hidden, non-transfer rows on or after `sinceDate` (and before `untilDate` when given). */
function baseWhere(sinceDate: string, untilDate?: string) {
  return and(
    eq(accounts.hidden, false),
    notTransfer,
    gte(transactions.localDate, sinceDate),
    untilDate ? lt(transactions.localDate, untilDate) : undefined,
  );
}

export function spendingByPeriod(db: Db, opts: { days: number; granularity: Granularity; today: string }): PeriodTotals[] {
  const since = addDays(opts.today, -opts.days);
  const period = periodExpr(opts.granularity);
  return db
    .select({ period, currency: accounts.currency, outflowCents: outflow, inflowCents: inflow })
    .from(transactions)
    .innerJoin(accounts, eq(transactions.accountId, accounts.id))
    .leftJoin(transactionMeta, eq(transactionMeta.transactionId, transactions.id))
    .where(baseWhere(since))
    .groupBy(period, accounts.currency)
    .orderBy(accounts.currency, period)
    .all();
}

export function topPayees(db: Db, opts: { days: number; today: string; limit?: number; sinceDate?: string }): { label: string; currency: string; outflowCents: number; count: number }[] {
  const since = opts.sinceDate ?? addDays(opts.today, -opts.days);
  const limit = opts.limit ?? 10;
  const label = sql<string>`coalesce(nullif(${transactions.payee}, ''), ${transactions.description})`;
  const outflowCents = sql<number>`sum(-${transactions.amountCents})`.mapWith(Number);
  const rows = db
    .select({ label, currency: accounts.currency, outflowCents, count: sql<number>`count(*)`.mapWith(Number) })
    .from(transactions)
    .innerJoin(accounts, eq(transactions.accountId, accounts.id))
    .leftJoin(transactionMeta, eq(transactionMeta.transactionId, transactions.id))
    .where(and(baseWhere(since), sql`${transactions.amountCents} < 0`))
    .groupBy(label, accounts.currency)
    .orderBy(accounts.currency, desc(outflowCents), label)
    .all();
  const perCurrencyCount = new Map<string, number>();
  const out: typeof rows = [];
  for (const r of rows) {
    const n = perCurrencyCount.get(r.currency) ?? 0;
    if (n >= limit) continue;
    perCurrencyCount.set(r.currency, n + 1);
    out.push(r);
  }
  return out;
}

export function spendingByCategory(db: Db, opts: { sinceDate: string; untilDate?: string }): { categoryId: number | null; name: string; currency: string; outflowCents: number; count: number }[] {
  const outflowCents = sql<number>`sum(-${transactions.amountCents})`.mapWith(Number);
  return db
    .select({
      categoryId: transactionMeta.categoryId,
      name: sql<string>`coalesce(${categories.name}, 'Uncategorised')`,
      currency: accounts.currency,
      outflowCents,
      count: sql<number>`count(*)`.mapWith(Number),
    })
    .from(transactions)
    .innerJoin(accounts, eq(transactions.accountId, accounts.id))
    .leftJoin(transactionMeta, eq(transactionMeta.transactionId, transactions.id))
    .leftJoin(categories, eq(categories.id, transactionMeta.categoryId))
    .where(and(baseWhere(opts.sinceDate, opts.untilDate), sql`${transactions.amountCents} < 0`))
    .groupBy(transactionMeta.categoryId, accounts.currency)
    .orderBy(accounts.currency, desc(outflowCents))
    .all()
    .map((r) => ({ ...r, categoryId: r.categoryId ?? null }));
}
```

```ts
import { addDays, todayLocal } from "../time";

export type SpendingCategory = { categoryId: number | null; name: string; currency: string; outflowCents: number };
export type SpendingPayload = { range: SpendingRange; days: SpendingDay[]; topPayees: ReturnType<typeof topPayees>; byCategory: SpendingCategory[] };

export function buildSpending(db: Db, range: SpendingRange, opts: { tz: string; nowMs?: number }): SpendingPayload {
  const nowMs = opts.nowMs ?? Date.now();
  const today = todayLocal(opts.tz, nowMs);
  const dayCount = range === "today" ? 1 : range === "7d" ? 7 : 30;
  const sinceDate = range === "today" ? today : addDays(today, -dayCount);
  const rows = spendingByPeriod(db, { days: dayCount, granularity: "day", today });
  const filtered = range === "today" ? rows.filter((r) => r.period === today) : rows;
  const days = filtered.map((r) => ({ date: r.period, currency: r.currency, outflowCents: r.outflowCents, inflowCents: r.inflowCents }));
  return {
    range,
    days,
    topPayees: topPayees(db, { days: dayCount, today, limit: 10, sinceDate }),
    byCategory: spendingByCategory(db, { sinceDate }).map(({ categoryId, name, currency, outflowCents }) => ({ categoryId, name, currency, outflowCents })),
```

```ts
import { setCategory, clearCategoryOverride, setTransfer } from "@/lib/queries/meta";

/** Only ever send the browser back to a path on this site. */
function safeReturn(raw: FormDataEntryValue | null): string {
  const s = String(raw ?? "");
  return s.startsWith("/") && !s.startsWith("//") ? s : "/transactions";
}

function revalidateAll(): void {
  revalidatePath("/");
  revalidatePath("/transactions");
  revalidatePath("/spending");
  revalidatePath("/accounts/[id]", "page");
}

export async function setTransactionCategory(formData: FormData): Promise<void> {
  const id = String(formData.get("id") ?? "");
  const raw = String(formData.get("category") ?? "");
  if (id) {
    const db = getDb();
    if (raw === "clear") clearCategoryOverride(db, id);
    else if (raw === "") setCategory(db, id, null);
    else if (/^\d+$/.test(raw)) setCategory(db, id, Number(raw));
    revalidateAll();
  }
  redirect(safeReturn(formData.get("returnTo")));
}

export async function setTransactionTransfer(formData: FormData): Promise<void> {
  const id = String(formData.get("id") ?? "");
  const flag = String(formData.get("transfer") ?? "");
  if (id && (flag === "1" || flag === "0")) {
    setTransfer(getDb(), id, flag === "1");
    revalidateAll();
  }
  redirect(safeReturn(formData.get("returnTo")));
}
```

```ts
import { compileRules } from "../tagging/rules";
import { accounts, transactions } from "../db/schema";
import { desc } from "drizzle-orm";
import { effectiveDate } from "./transactions";

export function previewRuleMatches(db: Db, input: RuleInput, limit = 5) {
  const { compiled } = compileRules([{ id: 0, field: input.field, match: input.match, pattern: input.pattern.trim(), accountId: input.accountId ?? null, minAmountCents: input.minAmountCents ?? null, maxAmountCents: input.maxAmountCents ?? null, categoryId: input.categoryId ?? null, markTransfer: input.markTransfer ?? false }]);
  const rule = compiled[0];
  if (!rule) return { count: 0, sample: [] };
  const rows = db
    .select({ id: transactions.id, accountId: transactions.accountId, localDate: transactions.localDate, payee: transactions.payee, description: transactions.description, amountCents: transactions.amountCents, currency: accounts.currency })
    .from(transactions)
    .innerJoin(accounts, eq(transactions.accountId, accounts.id))
    .orderBy(desc(effectiveDate), desc(transactions.id))
    .all();
  const matched = rows.filter((r) => rule.test(r));
  return { count: matched.length, sample: matched.slice(0, limit).map(({ accountId: _a, ...rest }) => rest) };
}
```
